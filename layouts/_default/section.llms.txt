{{- /* Section llms.txt, for every section type (the docs-type copy under
   layouts/docs/ was folded into this one).

   - Above a version tree (a product or docs root section): an INDEX, built by
     utils/llms-index-tree.html. Versions link to their own llms.txt, latest
     first; nothing is inlined.
   - A version root, anything inside a version tree, and a leaf section with no
     subsections: the FULL page listing, built by utils/llms-page-tree.html.

   "Version tree" comes from utils/version-root.html, not from counting URL
   segments, so whether the baseURL carries a path (the docs hub's /<product>/
   in production, none in a local build) no longer changes which branch a
   section takes. */ -}}
{{- $prodHost := partial "utils/prod-host.html" . -}}
{{- $v := partial "utils/version-root.html" . -}}
{{- $children := partial "utils/llms-children.html" . -}}
{{- $intro := .Description | replaceRE `\s+` " " | strings.TrimSpace -}}
{{- if and (not $intro) (eq .Type "docs") .Parent -}}
  {{- if .Parent.IsHome -}}
    {{- $intro = site.Params.llms.docsIntro | default (site.Params.description | default "") -}}
  {{- end -}}
{{- end -}}
{{- $body := "" -}}
{{- if and (not $v.isVersioned) $children -}}
  {{- /* This file is the section's own listing, so its own pages are listed in
         full before the subsection index. Without them, pages that sit next to
         a subsection in an unversioned tree were in no llms.txt at all. */ -}}
  {{- range where .RegularPages "Params.llms" "ne" false -}}
    {{- $body = print $body (partial "utils/llms-entry.html" (dict "page" . "prodHost" $prodHost)) "\n" -}}
  {{- end -}}
  {{- range $children -}}
    {{- $body = print $body (partial "utils/llms-index-tree.html" (dict "page" . "level" 2 "prodHost" $prodHost)) -}}
  {{- end -}}
{{- else -}}
  {{- $body = partial "utils/llms-page-tree.html" (dict "page" . "level" 2 "prodHost" $prodHost) -}}
{{- end -}}
# {{ .Title }}
{{ with $intro }}
> {{ . }}
{{ end }}
{{- with partial "utils/llms-href.html" . }}
For the full site index, see [llms.txt]({{ print $prodHost . }}).
{{ end }}
{{ $body | replaceRE `\n{3,}` "\n\n" | strings.TrimSpace }}
