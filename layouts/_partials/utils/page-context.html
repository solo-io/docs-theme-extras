{{/* page-context — return the rendering context for the current page. Returns a dict with: section the section / product name (e.g. "kubernetes", "test") version the version slug for the page (e.g. "v2", "main") condition the buildCondition string (drives `conditional-text`) prefix the URL prefix to prepend to relative paths in cards/links (e.g. "/docs/kubernetes/v2", "/test/v2") Dispatches on Site.Params.pageContextMode: "siteParams" — derives from explicit Site.Params (folder, currentProduct, buildCondition, versions). Used by multi-product hubs: the host owns the "//" path prefix so it doesn't appear in Page.RelPermalink. "url" — parses Page.RelPermalink as /docs/
//... (fallback /docs//...). Used by single-site repos where the URL itself encodes section and version. Auto-detection: if pageContextMode isn't set, the partial picks "siteParams" when Site.Params.folder is set (signals multi-product hub config) and "url" otherwise. Set the mode explicitly to override. */}} {{- $mode := .Site.Params.pageContextMode -}} {{- if not $mode -}} {{- if .Site.Params.folder -}} {{- $mode = "siteParams" -}} {{- else -}} {{- $mode = "url" -}} {{- end -}} {{- end -}} {{- $section := "" -}} {{- $version := "" -}} {{- $condition := "" -}} {{- $prefix := "" -}} {{- if eq $mode "siteParams" -}} {{- $folder := .Site.Params.folder | default "" -}} {{- /* A product with more than one parallel documentation section (site.Params.sections, see utils/section-segment.html) nests its version trees one level deeper, under a section segment: //
//…. .Page.FirstSection only returns the first path level under the product, which would resolve to the section instead of the version in that shape, so derive the version from the full URL instead when a section is present. */ -}} {{- $sectionSegInfo := partial "utils/section-segment.html" .Page -}} {{- $sectionSeg := $sectionSegInfo.segment -}} {{- if $sectionSeg -}} {{- $relParts := split (strings.Trim .Page.RelPermalink "/") "/" -}} {{- $versionIdx := add $sectionSegInfo.index 1 -}} {{- if gt (len $relParts) $versionIdx -}} {{- $version = index $relParts $versionIdx -}} {{- end -}} {{- else -}} {{- $sectionUrl := .Page.FirstSection.RelPermalink -}} {{- $version = trim (replace $sectionUrl (printf "/%s" $folder) "") "/" -}} {{- end -}} {{- $section = $folder -}} {{- $condition = .Site.Params.buildCondition | default "" -}} {{- /* The language segment sits between the product and the section (///
//), and $prefix is REBUILT from params rather than sliced off the page's own URL, so it silently dropped it. Every {{< card >}} on a translated page then linked to the English copy — 67 hrefs on one agentgateway ja build. LanguagePrefix is "" for the default language, so English output is byte-identical. */ -}} {{- $lang := .Site.LanguagePrefix | default "" -}} {{- if and $folder $version -}} {{- if $sectionSeg -}} {{- $prefix = printf "/%s%s/%s/%s" $folder $lang $sectionSeg $version -}} {{- else -}} {{- $prefix = printf "/%s%s/%s" $folder $lang $version -}} {{- end -}} {{- else if $folder -}} {{- $prefix = printf "/%s%s" $folder $lang -}} {{- end -}} {{- else -}} {{/* url mode */}} {{- $rel := trim .Page.RelPermalink "/" -}} {{- $parts := split $rel "/" -}} {{- if and (ge (len $parts) 2) (eq (index $parts 0) "docs") -}} {{- /* Known sections come from the keys of site.Params.sections (see utils/section-segment.html) instead of a hardcoded list, so each OSS-shaped site declares its own set by registering a table there — even an empty one — for each section it has (agentgateway: kubernetes/standalone; kgateway: envoy/agentgateway) rather than sharing one list across every consumer. */ -}} {{- $known := slice -}} {{- range $k, $_ := .Site.Params.sections -}} {{- $known = $known | append $k -}} {{- end -}} {{- if and (ge (len $parts) 3) (in $known (index $parts 1)) -}} {{- $section = index $parts 1 -}} {{- $version = index $parts 2 -}} {{- $condition = $section -}} {{- $prefix = printf "/docs/%s/%s" $section $version -}} {{- else -}} {{- $version = index $parts 1 -}} {{- $prefix = printf "/docs/%s" $version -}} {{- end -}} {{- end -}} {{- end -}} {{- return dict "section" $section "version" $version "condition" $condition "prefix" $prefix -}}
