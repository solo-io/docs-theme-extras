{{- /* Override: nil-safe version of hextra's extract-headings partial. Lives in _partials/ because hextra's _partials/ copy shadows partials/. Keys are plainified and unescaped so typographer entities such as ’ don't show as literal text in search results (fixed upstream in hextra v0.13.0). */ -}} {{- with .target -}} {{- with .Headings -}} {{- range $heading := . -}} {{- if and (eq $heading.Level 0) (not $heading.Title) -}} {{- $.scratch.Add "keys" (slice $heading.Title) -}} {{- else -}} {{- $headingTitle := $heading.Title | plainify | htmlUnescape | strings.TrimSpace -}} {{- $key := (printf "%s#%s" $heading.ID $headingTitle) -}} {{- $.scratch.Add "keys" (slice $key) -}} {{- end -}} {{- $title := (printf "%s" $heading.Level $heading.Title) | htmlUnescape -}} {{- $.scratch.Add "titles" (slice $title) -}} {{- partial "utils/extract-headings.html" (dict "target" $heading "scratch" $.scratch ) }} {{- end -}} {{- end -}} {{- end -}}
