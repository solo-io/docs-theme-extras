Usage: {{ $html = partial "utils/unhide-tabs.html" $html }} Input/output is HTML as a string (or template.HTML); the caller marks the result safeHTML again if re-emitting as markup. */ -}} {{- $html := . -}} {{- $html = $html | replaceRE `(?s)]*role="tablist"[^>]*>.*?` "" -}} {{- $html = $html | replaceRE `aria-hidden="true"` "" -}} {{- $html = $html | replaceRE `
]*aria-labelledby="([^"]*)"[^>]*>` `

Option: $1

` -}} {{- return $html -}}
