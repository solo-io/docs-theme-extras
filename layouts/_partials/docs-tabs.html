{{- /* docs-tabs.html — tab-navigation band (issue solo-io/docs#3164). Renders a horizontal row of tabs BETWEEN the top navbar and the sidebar/content row. Each tab scopes the left nav to one top-level content directory of the current version, the way Mintlify / Portkey group concepts so the left nav stays short. Left-aligned, its own band (Portkey style). Model (directory-based): - Config declares the tab bar: an ordered `[[params.docTabs]]` list, each with a `name` (display label), an `id` (a top-level content directory under the version root), and optional `default = true`. This is version-agnostic (the labels rarely change), so one block per product covers every version. - A tab owns every page in its `id` directory. The sidebar (sidebar.html) roots the left nav INSIDE that directory, so the directory's own node never appears — it *is* the tab. Content is fully partitioned by directory, one dir per tab (the default tab gets a dir too). Opt-in: - No `params.docTabs` → this partial emits nothing and records "disabled" in the page store, so the sidebar renders the whole version tree as if the feature didn't exist. - Fewer than 2 present tab directories in the current version → same as disabled. A single-tab band would add chrome without slimming anything, and it lets a version that hasn't been organised into tabs keep the plain full-tree nav. Sharing with the sidebar: We compute the active tab here and stash it — plus the enabled flag, the active tab's directory id, and the ordered tab list — on the page store. The sidebar reads it back and roots its tree at the active tab's directory, so the version-root detection lives in one place per render. single.html / list.html call this partial BEFORE the sidebar, so the store is populated first. Version-root detection is shared with the sidebar via utils/version-root.html, so both agree on which version tree the page is in. */ -}} {{- $context := . -}} {{- $tabsCfg := site.Params.docTabs -}} {{- /* Default: feature off. Overwritten below once we confirm >=2 tabs. */ -}} {{- $context.Store.Set "docTabs" (dict "enabled" false) -}} {{- if $tabsCfg -}} {{- /* Resolve the version root through the shared helper (same detection the sidebar uses), so the band and the left nav never disagree about the active version tree. */ -}} {{- $vr := partial "utils/version-root.html" $context -}} {{- if $vr.isVersioned -}} {{- $versionRoot := $vr.docsSection -}} {{- if $versionRoot -}} {{- /* $lookupPath is the version root relative to contentDir (the shape version-root.html already normalises), so site.GetPage resolves a tab's directory the same way in OSS, enterprise, and local dev. */ -}} {{- $lookupPath := $vr.lookupPath -}} {{- /* ---- Default tab name: active when the page sits above every tab directory (e.g. the version-root landing). ------------------ */ -}} {{- $defaultTab := (index $tabsCfg 0).name -}} {{- range $tabsCfg -}} {{- if .default -}}{{- $defaultTab = .name -}}{{- end -}} {{- end -}} {{- /* ---- Active tab: the tab whose directory is the current page or an ancestor of it (Page relationships, not URL strings, so the OSS / enterprise / local-dev URL shapes all work). ---------- */ -}} {{- $activeTab := $defaultTab -}} {{- range $tabsCfg -}} {{- $name := .name -}} {{- with .id -}} {{- with site.GetPage (printf "%s%s" $lookupPath .) -}} {{- if or (eq .RelPermalink $context.RelPermalink) ($context.IsDescendant .) -}} {{- $activeTab = $name -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* ---- Ordered list of present tabs (those whose `id` directory exists). Each tab links to its directory landing. A tab with no `id`, or whose directory is missing, is skipped. ------------- */ -}} {{- $tabs := slice -}} {{- range $tabsCfg -}} {{- $name := .name -}} {{- $id := .id -}} {{- with $id -}} {{- with site.GetPage (printf "%s%s" $lookupPath .) -}} {{- $tabs = $tabs | append (dict "name" $name "href" .RelPermalink "id" $id) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Resolve the active tab's directory id from the present-tab list (so a configured-but-missing tab can't win) for the sidebar to root at. */ -}} {{- $activeId := "" -}} {{- range $tabs -}}{{- if eq .name $activeTab -}}{{- $activeId = .id -}}{{- end -}}{{- end -}} {{- /* Share with the sidebar regardless of count so it can decide. The tab list travels too, so the sidebar can render the mobile drawer's tab-chip row (the band itself is hidden below the desktop-sidebar breakpoint). `activeId` is the directory the sidebar roots at. */ -}} {{- $enabled := ge (len $tabs) 2 -}} {{- $context.Store.Set "docTabs" (dict "enabled" $enabled "active" $activeTab "activeId" $activeId "tabs" $tabs) -}} {{- if $enabled -}} {{- /* The full-bleed band centers its tabs inside the SAME page-width container the content column uses (utils/page-width), so the tab row's left edge lines up with the sidebar/content instead of the viewport edge. Using the shared partial keeps them aligned for every page.width (wide / full / default). */ -}}
{{- range $tabs -}} {{- $isActive := eq .name $activeTab -}} {{ .name }} {{- end -}}
