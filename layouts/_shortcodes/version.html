Mirrors the same matching in the Python doc-test extractor (_evaluate_version_block). */ -}} {{- if ne $condition "" -}} {{- range $versions -}} {{- if and (in $condition .linkVersion) (not $emitted) -}} {{- $productMatches := or (not $currentProduct) (eq .product $currentProduct) -}} {{- if $productMatches -}} {{- if partial "utils/gate-decide.html" (dict "sc" $ "tokens" (slice .version .linkVersion)) -}} {{- partial "utils/gate-emit.html" (dict "sc" $ "inner" $.Inner) -}} {{- $emitted = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* KEEPVERSION LANDING ZONE: no version resolved from the URL, but reuse.html injected a `version="…"` param naming the parent's version. PRESERVED QUIRK, deliberately: an `exclude-if` block here emits nothing when `version=` does not resolve. In the old code that fell out of a substring test against an empty string; it is explicit now. There is no fixture for it, so it is kept as-is rather than "fixed" blind — changing it would silently start emitting content on every unresolvable enterprise-only page. */ -}} {{- if and (eq $condition "") (not $emitted) -}} {{- $versionImport := .Get "version" -}} {{- $versionAdjusted := "" -}} {{- range $versions -}} {{- if not $versionAdjusted -}} {{- if or (eq .version $versionImport) (eq .linkVersion $versionImport) -}} {{- $versionAdjusted = .version -}} {{- else if and .ossDir (eq .ossDir $versionImport) -}} {{- $versionAdjusted = .version -}} {{- end -}} {{- end -}} {{- end -}} {{- if $versionAdjusted -}} {{- range $versions -}} {{- if and (eq .version $versionAdjusted) (not $emitted) -}} {{- $productMatches := or (not $currentProduct) (eq .product $currentProduct) -}} {{- if $productMatches -}} {{- if partial "utils/gate-decide.html" (dict "sc" $ "tokens" (slice .version .linkVersion)) -}} {{- partial "utils/gate-emit.html" (dict "sc" $ "inner" $.Inner) -}} {{- $emitted = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
